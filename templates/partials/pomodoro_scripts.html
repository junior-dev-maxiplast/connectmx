}); } startBtn.addEventListener("click", () => { if (isRunning) { stopTimer(); renderModeChrome(); return; } isRunning = true; timerId = setInterval(tick, 1000); renderModeChrome(); }); resetBtn.addEventListener("click", () => { stopTimer();
